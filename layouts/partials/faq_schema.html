{{- /* Emits FAQPage structured data for posts whose content is a list of
QuestionAnswer
blocks, so search engines can understand and potentially surface individual Q&A pairs. */ -}} {{- if eq .Kind "page" }} {{- $chunks := findRE `(?s).*?.*?` .Content }} {{- if ge (len $chunks) 2 }} {{- $faq := slice }} {{- range $chunks }} {{- $chunk := . }} {{- $summaryHTML := replaceRE `(?s)^(.*?).*$` "$1" $chunk }} {{- $answerHTML := replaceRE `(?s)^.*?(.*)$` "$1" $chunk }} {{- $titleHTML := replaceRE `(?s)^(.*?).*$` "$1" $summaryHTML }} {{- $question := $titleHTML | plainify | htmlUnescape | htmlUnescape | strings.TrimSpace }} {{- $answer := $answerHTML | plainify | strings.TrimSpace }} {{- $answer = strings.Truncate 300 $answer | htmlUnescape | htmlUnescape }} {{- if and $question $answer }} {{- $faq = $faq | append (dict "@type" "Question" "name" $question "acceptedAnswer" (dict "@type" "Answer" "text" $answer)) }} {{- end }} {{- end }} {{- if gt (len $faq) 0 }} {{- end }} {{- end }} {{- end }}
